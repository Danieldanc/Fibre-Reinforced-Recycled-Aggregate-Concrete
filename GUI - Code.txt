# Tuned Gradient Boosting hyper-parameters (from the study's hyper-parameter search)
GB_PARAMS = dict(n_estimators=600, max_depth=7, learning_rate=0.05,
                 subsample=0.85, min_samples_leaf=5, random_state=42)

def train_model():
    """Train Gradient Boosting on the embedded data and return (deploy_model, metrics)."""
    arr = np.array(DATA, dtype=float)
    X, y = arr[:, :7], arr[:, 7]

    # 80 / 20 split (seed 42) -> reproduces the reported metrics
    Xtr, Xte, ytr, yte = train_test_split(X, y, test_size=0.20, random_state=42)
    split_model = GradientBoostingRegressor(**GB_PARAMS)
    split_model.fit(Xtr, ytr)

    def trio(a, b):
        return (r2_score(a, b),
                float(np.sqrt(mean_squared_error(a, b))),
                mean_absolute_error(a, b))

    tr = trio(ytr, split_model.predict(Xtr))
    te = trio(yte, split_model.predict(Xte))
    metrics = {"train_r2": tr[0], "train_rmse": tr[1], "train_mae": tr[2],
               "test_r2": te[0], "test_rmse": te[1], "test_mae": te[2]}

    # Final deployment model: refit on ALL 360 specimens for best predictions
    deploy = GradientBoostingRegressor(**GB_PARAMS)
    deploy.fit(X, y)
    return deploy, metrics


def strength_class(fc):
    """Indicative residual-strength class for the studied FRRAC system."""
    if fc < 10.0:
        return "Severe Degradation  (< 10 MPa)"
    elif fc < 18.0:
        return "High Degradation  (10 - 18 MPa)"
    elif fc < 28.0:
        return "Moderate Residual Strength  (18 - 28 MPa)"
    else:
        return "Low Degradation / Near-Ambient  (> 28 MPa)"

    # ---------------- actions ----------------
    def calculate(self):
        try:
            x = [float(self.entries[name].get()) for (_, name, _, _) in FEATURES]
        except ValueError:
            messagebox.showerror("Invalid input",
                                 "Please enter a valid number in every field.")
            return
        fc = float(self.model.predict(np.array([x]))[0])
        self.pred_var.set(f"{fc:.2f} MPa")
        self.class_var.set(strength_class(fc))

    def reset(self):
        for (_, name, _, _) in FEATURES:
            self.entries[name].delete(0, "end")
        self.pred_var.set("--")
        self.class_var.set("--")

    def load_example(self):
        for (_, name, _, default) in FEATURES:
            self.entries[name].delete(0, "end")
            self.entries[name].insert(0, str(default))
        self.calculate()
